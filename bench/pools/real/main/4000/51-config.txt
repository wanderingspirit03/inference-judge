=== scripts/backup.sh ===

#!/bin/bash
# Nightly backup for the food bank database
# cron: 0 2 * * * /srv/foodbank/scripts/backup.sh

BACKUP_DIR=/srv/backups
DATE=`date +%d-%m-%Y`
cd /srv/foodbank

docker exec foodbank_db_1 pg_dump -U foodbank foodbank > $BACKUP_DIR/db_$DATE.sql

tar -czf $BACKUP_DIR/uploads_$DATE.tar.gz uploads

# remove old backups
find $BACKUP_DIR -name "*.sql" -mtime +7 -exec rm {} \;
find $BACKUP_DIR -name "*.tar.gz" -mtime +7 -exec rm {} \;

echo "Backup finished $DATE" >> /var/log/foodbank_backup.log

=== scripts/restore.sh ===

#!/bin/bash
# usage: restore.sh <date>
DATE=$1
BACKUP_DIR=/srv/backups

cd /srv/foodbank
docker-compose down
docker volume rm foodbank_dbdata
docker-compose up -d db
sleep 5
cat $BACKUP_DIR/db_$DATE.sql | docker exec -i foodbank_db_1 psql -U foodbank foodbank
tar -xzf $BACKUP_DIR/uploads_$DATE.tar.gz -C /srv/foodbank
docker-compose up -d
echo done

=== scripts/healthcheck.sh ===

#!/bin/bash
# Check the site is up every 5 minutes: */5 * * * * /srv/foodbank/scripts/healthcheck.sh
URL=http://localhost:3000/health
CODE=$(curl -s -o /dev/null -w "%{http_code}" $URL)
if [ $CODE != 200 ]; then
  echo "Site down at `date`" | mail -s "foodbank DOWN" coordinator@example.org
  docker restart foodbank_web_1
fi
df -h / | tail -1 | awk '{ if ($5+0 > 90) print "disk almost full: " $5 }' | mail -s "foodbank disk" coordinator@example.org

=== scripts/db-init.sh ===

#!/bin/bash
# Create the schema on a new database and load the starter data
set -x
until docker exec foodbank_db_1 pg_isready -U foodbank; do
  sleep 1
done
docker exec -i foodbank_db_1 psql -U foodbank foodbank < db/schema.sql
docker exec -i foodbank_db_1 psql -U foodbank foodbank < db/seed.sql
for f in db/migrations/*.sql; do
  echo "applying $f"
  docker exec -i foodbank_db_1 psql -U foodbank foodbank < $f
done
echo "UPDATE users SET password='admin' WHERE email='coordinator@example.org';" | docker exec -i foodbank_db_1 psql -U foodbank foodbank

#!/bin/bash
# Certificates are issued by hand twice a year. Run this to renew.

=== Notes from the previous maintainer (README excerpt) ===

- The app holds names, phone numbers and addresses of about 1,400 households who receive food parcels, plus around 90 volunteers and their availability. It is run by a small charity with no paid staff.
- Backups go to the same server. Once, a few years ago, the server disk filled and the app was down for a weekend.
- The rota emails go out on Thursday at 17:00 for the weekend shifts and the cron for that is inside the application container.
- We deploy by pushing to main. Sometimes people push to develop by mistake and it deploys too.
- Uploaded delivery photos can include house fronts, and sometimes people's faces.
- If the server dies, the plan is "Dana will rebuild it from this repo".
