#!/bin/bash
# Deploys the callout service to the base station server.
# Usage: ./deploy.sh

cd /home/rescue/callout
git pull origin master

docker stop callout
docker rm callout
docker build -t callout .
docker run -d --name callout -p 5000:5000 \
  -v /home/rescue/callout/data:/app/data \
  callout

echo "Deployed at $(date)" >> deploy.log

# Quick check that the service is up
sleep 2
curl http://localhost:5000/health || echo "health check failed"

# Back up the phone list
cp data/phones.csv /tmp/phones_backup_$(date +%Y%m%d).csv
chmod 777 /tmp/phones_backup_*.csv

# Tell the team lead that a deploy happened
curl -X POST "https://hooks.example.org/notify?token=9f8e7d6c5b4a" \
  -d "text=callout deployed on $(hostname)"
