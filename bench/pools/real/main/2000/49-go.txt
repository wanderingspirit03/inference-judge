// Package bakeplan turns tomorrow's sales forecast and today's leftover
// stock into a list of oven batches for one shop of the Crumbline Bakery
// chain.
package bakeplan

import (
	"errors"
	"fmt"
	"sort"
	"strings"
	"time"
)

// Product describes one item the bakery makes.
type Product struct {
	SKU            string
	Name           string
	UnitsPerTray   int
	BakeMinutes    int
	ShelfLifeHours int
	MinTrays       int // smallest batch worth firing up an oven for
}

// Line is a quantity of one product, used for both forecasts and stock.
type Line struct {
	SKU   string
	Units int
}

// Oven is one deck oven in a shop.
type Oven struct {
	Name         string
	TraysPerLoad int
}

// Batch is one oven load.
type Batch struct {
	SKU   string
	Trays int
	Oven  string
	Start time.Time
	End   time.Time
}

// Plan is the output for one shop and one day.
type Plan struct {
	Shop      string
	Day       time.Time
	Batches   []Batch
	Shortfall map[string]int
}

// ErrNoOven is returned when a shop has no ovens configured.
var ErrNoOven = errors.New("bakeplan: no ovens configured")

// Catalogue maps SKUs to products.
type Catalogue map[string]Product

// NewCatalogue indexes a list of products by SKU.
func NewCatalogue(products []Product) Catalogue {
	c := make(Catalogue, len(products))
	for _, p := range products {
		c[p.SKU] = p
	}
	return c
}

// Lookup finds a product by SKU, ignoring case.
func (c Catalogue) Lookup(sku string) (Product, bool) {
	p, ok := c[strings.ToUpper(sku)]
	return p, ok
}

// TraysNeeded returns how many whole trays must be baked to cover the
// demand after using what is already in stock.
func TraysNeeded(demand, stock, unitsPerTray int) int {
	need := demand - stock
	if need <= 0 {
		return 0
	}
	return (need + unitsPerTray - 1) / unitsPerTray
}

type job struct {
	product Product
	trays   int
}

// Build plans the day's baking. Longer bakes are placed first, and each
// load goes into whichever oven becomes free earliest.
func Build(shop string, day time.Time, cat Catalogue, forecast, stock []Line,
	ovens []Oven, firstLoad time.Time) (Plan, error) {

	if len(ovens) == 0 {
		return Plan{}, ErrNoOven
	}

	stockBySKU := map[string]int{}
	for _, s := range stock {
		stockBySKU[s.SKU] += s.Units
	}

	plan := Plan{Shop: shop, Day: day}
	var jobs []job

	for _, f := range forecast {
		p, ok := cat.Lookup(f.SKU)
		if !ok {
			return Plan{}, fmt.Errorf("bakeplan: unknown SKU %q", f.SKU)
		}
		trays := TraysNeeded(f.Units, stockBySKU[f.SKU], p.UnitsPerTray)
		if trays == 0 {
			continue
		}
		if trays < p.MinTrays {
			plan.Shortfall[f.SKU] = trays * p.UnitsPerTray
			continue
		}
		jobs = append(jobs, job{product: p, trays: trays})
	}

	sort.Slice(jobs, func(i, j int) bool {
		return jobs[i].product.BakeMinutes > jobs[j].product.BakeMinutes
	})

	free := make([]time.Time, len(ovens))
	for i := range free {
		free[i] = firstLoad
	}

	for _, j := range jobs {
		remaining := j.trays
		for remaining > 0 {
			k := earliest(free)
			load := ovens[k].TraysPerLoad
			if load > remaining {
				load = remaining
			}
			start := free[k]
			end := start.Add(time.Duration(j.product.BakeMinutes) * time.Minute)
			plan.Batches = append(plan.Batches, Batch{
				SKU:   j.product.SKU,
				Trays: load,
				Oven:  ovens[k].Name,
				Start: start,
				End:   end,
			})
			free[k] = end
			remaining -= load
		}
	}
	return plan, nil
}

// earliest returns the index of the oven that is free soonest.
func earliest(free []time.Time) int {
	best := 0
	for i := 1; i < len(free); i++ {
		if free[i].Before(free[best]) {
			best = i
		}
	}
	return best
}

// TotalTrays adds up the trays in every batch.
func (p Plan) TotalTrays() int {
	n := 0
	for _, b := range p.Batches {
		n += b.Trays
	}
	return n
}

// FinishTime is when the last batch comes out.
func (p Plan) FinishTime() time.Time {
	var last time.Time
	for _, b := range p.Batches {
		if b.End.After(last) {
			last = b.End
		}
	}
	return last
}

// Late lists the batches that come out after the shop opens.
func (p Plan) Late(opening time.Time) []Batch {
	var late []Batch
	for _, b := range p.Batches {
		if b.End.After(opening) {
			late = append(late, b)
		}
	}
	return late
}

// Expiry is when a batch must be taken off the shelf.
func Expiry(b Batch, cat Catalogue) time.Time {
	p := cat[b.SKU]
	return b.End.Add(time.Duration(p.ShelfLifeHours) * time.Hour)
}

// Summary renders the plan as a short text for the shop screen.
func (p Plan) Summary() string {
	var sb strings.Builder
	fmt.Fprintf(&sb, "%s %s\n", p.Shop, p.Day.Format("2006-01-02"))
	for _, b := range p.Batches {
		fmt.Fprintf(&sb, "%s %s x%d %s-%s\n", b.Oven, b.SKU, b.Trays,
			b.Start.Format("15:04"), b.End.Format("15:04"))
	}
	return sb.String()
}
